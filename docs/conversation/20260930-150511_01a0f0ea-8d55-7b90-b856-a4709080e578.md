1. **DNSの全ゾーンをメモリに保持し、問合せ時のSQLをなくす**

  for i,r:=range batch { results[i].Serial=d.serial;if err!=nil && results[i].Error=="" { results[i].Error=err.Error() };r.done<-results[i] }
 }
}

func (d *dnsControl) handler(allowed map[string]bool) http.Handler {
 mux:=http.NewServeMux()
 mux.HandleFunc("/",func(w http.ResponseWriter,r *http.Request) {
  host,_,_:=net.SplitHostPort(r.RemoteAddr)
  if !allowed[host] { http.Error(w,"forbidden",http.StatusForbidden);return }
  if r.Method!=http.MethodPost { http.Error(w,"POST required",http.StatusMethodNotAllowed);return }
  var update dnsUpdate
  if r.URL.Path=="/register" {
   if err:=json.NewDecoder(io.LimitReader(r.Body,1024)).Decode(&update);err!=nil || !validDNSUser(update.Name) { http.Error(w,"invalid DNS name",400);return }
   update.Reset=false
  } else if r.URL.Path=="/reset" { update.Reset=true } else { http.NotFound(w,r);return }
  update.done=make(chan dnsReply,1)
  select { case d.queue<-update:case <-r.Context().Done():return }
  select {
  case reply:=<-update.done:
   w.Header().Set("Content-Type","application/json")
   if reply.Error!="" { w.WriteHeader(http.StatusInternalServerError) }
   _=json.NewEncoder(w).Encode(reply)
  case <-r.Context().Done():
  }
 })
 return mux
}

func reloadDNSZone(serial uint32) error {

  }
